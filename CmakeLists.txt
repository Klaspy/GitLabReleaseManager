cmake_minimum_required(VERSION 3.16)

project(GitLabReleaseManager VERSION 0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt6 REQUIRED COMPONENTS Quick Network sql)

qt_standard_project_setup(REQUIRES 6.8)

set(app_resource_windows "${CMAKE_CURRENT_SOURCE_DIR}/GitLabReleaseManager.rc")

qt_add_executable(GitLabReleaseManager
    main.cpp
    src/projects/project.h
    src/projects/project.cpp
    src/projects/projectslistmodel.h
    src/projects/projectslistmodel.cpp
    src/projects/releaselistmodel.h
    src/projects/releaselistmodel.cpp
    src/databaseworker.h
    src/databaseworker.cpp
    src/global.h
    src/privatekeymodel.h
    src/privatekeymodel.cpp
    src/requester.h
    src/requester.cpp

    ${app_resource_windows}
)

qt_add_resources(GitLabReleaseManager "images"
PREFIX "/"
FILES
    images/reload.png
)

qt_add_qml_module(GitLabReleaseManager
    URI GitLabReleaseManagerModule
    VERSION 1.0
    RESOURCE_PREFIX /
    QML_FILES
        qml/App.qml
        qml/components/HeaderButton.qml
        qml/modal/AddProjectDialog.qml
        qml/modal/PreferencesDialog.qml
        qml/modal/PrivateKeysListDialog.qml
        qml/modal/ReleaseDialog.qml
        qml/stackelements/ProjectStackElement.qml
        qml/stackelements/ReleaseColumnElement.qml
        qml/stackelements/ReleasesStackElement.qml

    SOURCES
        src/projects/project.h
        src/projects/projectslistmodel.h
        src/projects/releaselistmodel.h
        src/databaseworker.h
        src/global.h
        src/privatekeymodel.h
)

add_subdirectory(CustomStyle)

# Qt for iOS sets MACOSX_BUNDLE_GUI_IDENTIFIER automatically since Qt 6.1.
# If you are developing for iOS or macOS you should consider setting an
# explicit, fixed bundle identifier manually though.
set_target_properties(GitLabReleaseManager PROPERTIES
#    MACOSX_BUNDLE_GUI_IDENTIFIER com.example.appcmakeQuickTest
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
    MACOSX_BUNDLE TRUE
    WIN32_EXECUTABLE TRUE
)

target_link_libraries(GitLabReleaseManager
    PRIVATE Qt6::Quick Qt6::Sql Qt6::Network
)

include_directories(
    src/
    src/projects
)

include(GNUInstallDirs)
install(TARGETS GitLabReleaseManager
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)
